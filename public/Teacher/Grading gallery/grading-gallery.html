function calculateTotal(items) {
    let total = 0;
    for(let i = 0; i < items.length; i++) {
        total += items[i].price * items[i].quantity;
    }
    return total;
}

const cart = [
    { id: 1, name: "Sách HTML", price: 50000, quantity: 2 },
    { id: 2, name: "Khóa học JS", price: 200000, quantity: 1 }
];

console.log("Tổng tiền:", calculateTotal(cart));
